function(vellum_target target)
    target_compile_features(${target} PUBLIC cxx_std_23)
    target_include_directories(${target} PUBLIC ${PROJECT_SOURCE_DIR}/include ${PROJECT_BINARY_DIR}/generated)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Werror)
        target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:-O3>)
        if(VELLUM_COVERAGE)
            target_compile_options(${target} PRIVATE --coverage -O0 -g)
            target_link_options(${target} PUBLIC --coverage)
        endif()
        if(VELLUM_SANITIZER)
            if(NOT VELLUM_SANITIZER MATCHES "^(address|undefined|thread)$")
                message(FATAL_ERROR "Unsupported VELLUM_SANITIZER")
            endif()
            target_compile_options(${target} PRIVATE -fsanitize=${VELLUM_SANITIZER} -fno-omit-frame-pointer -g)
            target_link_options(${target} PUBLIC -fsanitize=${VELLUM_SANITIZER})
        endif()
    endif()
endfunction()
