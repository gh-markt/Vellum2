cmake_minimum_required(VERSION 3.25)
project(vellum VERSION 0.34 LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(VELLUM_CORE_ONLY "Build dependency-independent core and IPC tests only" OFF)
option(VELLUM_PREPARE_ONLY "Configure only the explicit dependency preparation target" OFF)
option(VELLUM_COVERAGE "Instrument project code for gcov coverage" OFF)
set(VELLUM_SANITIZER "" CACHE STRING "address, undefined or thread (one build per sanitizer)")
set(VELLUM_DEPS_PREFIX "${PROJECT_SOURCE_DIR}/.deps/install" CACHE PATH "Local static dependency prefix")
set(VELLUM_DEPS_WORK "${PROJECT_SOURCE_DIR}/work/dependencies" CACHE PATH "Dependency work area")
set(VELLUM_DEPENDENCY_JOBS auto CACHE STRING "Dependency build jobs: auto or a positive integer")
set(VELLUM_MUPDF_ROOT "" CACHE PATH "Use an existing compatible static MuPDF installation")
set(VELLUM_V8_ROOT "" CACHE PATH "Use an existing compatible static V8 installation")
add_custom_target(package
    COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/scripts/package_source.py
        --root ${PROJECT_SOURCE_DIR} --output ${PROJECT_BINARY_DIR}/vellum-${PROJECT_VERSION}.zip
    USES_TERMINAL VERBATIM)
if(BUILD_TESTING)
    add_test(NAME testing.source_package COMMAND ${Python3_EXECUTABLE}
        ${PROJECT_SOURCE_DIR}/tests/test_package_source.py)
endif()
add_custom_target(prepare
    COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/scripts/prepare_dependencies.py
        --prefix ${VELLUM_DEPS_PREFIX} --work ${VELLUM_DEPS_WORK}
        --jobs ${VELLUM_DEPENDENCY_JOBS}
        --mupdf-root=${VELLUM_MUPDF_ROOT} --v8-root=${VELLUM_V8_ROOT}
    USES_TERMINAL VERBATIM)
if(VELLUM_PREPARE_ONLY)
    return()
endif()
find_package(Threads REQUIRED)
function(vellum_target target)
    target_compile_features(${target} PUBLIC cxx_std_23)
    target_include_directories(${target} PUBLIC ${PROJECT_SOURCE_DIR}/include ${PROJECT_BINARY_DIR}/generated)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Werror)
        target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:-O3>)
        if(VELLUM_COVERAGE)
            target_compile_options(${target} PRIVATE --coverage -O0 -g)
            target_link_options(${target} PUBLIC --coverage)
        endif()
        if(VELLUM_SANITIZER)
            if(NOT VELLUM_SANITIZER MATCHES "^(address|undefined|thread)$")
                message(FATAL_ERROR "Unsupported VELLUM_SANITIZER")
            endif()
            target_compile_options(${target} PRIVATE -fsanitize=${VELLUM_SANITIZER} -fno-omit-frame-pointer -g)
            target_link_options(${target} PUBLIC -fsanitize=${VELLUM_SANITIZER})
        endif()
    endif()
endfunction()
